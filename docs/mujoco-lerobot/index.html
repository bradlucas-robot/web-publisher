📦 第一步：Python 环境

cd /root/.openclaw/workspace
python3 -m venv lerobot_env
source lerobot_env/bin/activate
pip install --upgrade pip

🔧 第二步：安装 MuJoCo

source /root/.openclaw/workspace/lerobot_env/bin/activate
pip install mujoco mujoco-py
# 验证
python -c "import mujoco; print(mujoco.__version__)"

🤖 第三步：安装 LeRobot

source /root/.openclaw/workspace/lerobot_env/bin/activate
pip install lerobot --no-deps
pip install torch==2.10.0+cpu torchvision==0.25.0+cpu \
    --index-url https://download.pytorch.org/whl/cpu
pip install accelerate datasets deepdiff wandb tensorboard

🎮 第四步：安装 gym-pusht

apt-get install -y libgl1-mesa-glx libglib2.0-0
pip install gym-pusht

# 验证
python -c "import gym_pusht; print('✅ OK')"

🏋️ 第五步：训练 PPO 策略

from stable_baselines3 import PPO

model = PPO(
    policy="MlpPolicy",
    env=gym.make("gym_pusht/PushT-v0"),
    learning_rate=3e-4,
    n_steps=512,
    batch_size=64,
    gamma=0.99,
    verbose=1
)

model.learn(total_timesteps=50000, progress_bar=True)
model.save("ppo_pusht")
